Mise en route (5 minutes)
-------------------------
  # 1) pipeline vierge + hook (Branch A) — sorties bit-identiques
  git clone https://github.com/FBerthommier/covtl-pipeline.git
  cd covtl-pipeline
  git apply patch/covtl-pipeline-sandhi-hook.patch
  pip install -e .

  # 2) activer les règles — AU CHOIX :
  #    (le dossier du stub existe déjà : on copie le CONTENU, notez le "/.")
  cp -r sandhi/. vtl_synth/utils/sandhi/     # copie (recommandé)
  # ou :  pip install .                     # package pip

  # 3) vérifier
  vtl-synth run "faut il"    -l fr --sandhi standard --no-video -o out
  #  -> fo.til
  vtl-synth run "far away"   -l en --sandhi standard --no-video -o out
  #  -> fa.R@.we      (linking-r : NOUVEAU, l'anglais)
  vtl-synth run "los amigos" -l es --sandhi standard --no-video -o out
  #  -> lo.sa.mi.gos
  vtl-synth run "os amigos"  -l pt --sandhi standard --no-video -o out
  #  -> o.Z@.mi.guS
  vtl-synth run "a casa"     -l it --sandhi standard --no-video -o out
  #  -> a.kka.sa
  vtl-synth run "ein Apfel"  -l de --sandhi standard --no-video -o out
  #  -> ajn a.pf@l    (frontière glottale)

  # tests du package (autonomes)
  pip install pytest && python -m pytest tests/ -q     # 38 passed
